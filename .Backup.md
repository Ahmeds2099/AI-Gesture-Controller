import sys
import os

# --- 1. SILENCE TENSORFLOW LOGS (Must be at the very top) ---
os.environ['TF_CPP_MIN_LOG_LEVEL'] = '3' 

import cv2
import numpy as np
import pyautogui
import math
import time

# --- 2. COMPATIBILITY BRIDGE ---
np.float = float
np.int = int
if not hasattr(np, 'bool_'): np.bool_ = bool 

import mediapipe as mp
from mediapipe.tasks import python as tasks
from mediapipe.tasks.python import vision

def print_instructions():
    print("\n" + "="*60)
    print("         >> PRECISION GESTURE CONTROLLER v2.0 ACTIVE")
    print("="*60)
    print("\n[MOUSE CONTROL]")
    print("  • MOVE: Point with INDEX finger.")
    print("  • FREEZE (Brake): Stretch THUMB out slightly.")
    print("  • CLICK: While Frozen, PINCH Index and Thumb together.")
    print("\n[GRAB & DRAG]")
    print("  • TRIGGER: Quickly close your hand into a FIST (Open -> Close).")
    print("  • ACTION: Point -> Stretch Thumb (Freeze) -> Pinch to Grab.")
    print("\n[SPECIAL MODES]")
    print("  • SCROLL: Hold 2 fingers up (0.8s).")
    print("  • VOLUME: Hold 3 fingers up (0.8s).")
    print("\n[SAFETY]")
    print("  • RESET: Spread ALL 4 FINGERS wide to return to Mouse Mode.")
    print("  • EXIT: Press 'Q' on the video window to quit.")
    print("="*60 + "\n")

# --- 3. INITIAL SETTINGS ---
screen_w, screen_h = pyautogui.size()
pyautogui.PAUSE = 0
pyautogui.FAILSAFE = False

smoothening = 5
plocX, plocY = 0, 0
dragging, clicked = False, False
current_mode = 0 # 0:Mouse, 1:Scroll, 2:Volume, 3:Grab

gesture_start_time, last_detected_up_count = 0, -1
REQUIRED_HOLD_TIME, BRAKE_GRACE_PERIOD = 0.8, 0.8
brake_lock_expiry, last_open_time = 0, 0
THUMB_STRETCH_THRESHOLD = 0.055 
grab_pulse_ready = False

# --- 4. AI INITIALIZATION ---
model_path = os.path.join(os.getcwd(), 'hand_landmarker.task')
if not os.path.exists(model_path):
    print(f"FATAL ERROR: 'hand_landmarker.task' not found in {os.getcwd()}")
    sys.exit()

base_options = tasks.BaseOptions(model_asset_path=model_path)
options = vision.HandLandmarkerOptions(
    base_options=base_options,
    running_mode=vision.RunningMode.VIDEO,
    num_hands=1
)
detector = vision.HandLandmarker.create_from_options(options)

# Manual Drawing Logic
def draw_landmarks_manual(image, landmarks, color):
    connections = [
        (0,1), (1,2), (2,3), (3,4), (0,5), (5,6), (6,7), (7,8),
        (5,9), (9,10), (10,11), (11,12), (9,13), (13,14), (14,15), (15,16),
        (13,17), (17,18), (18,19), (19,20), (0,17)
    ]
    h, w, _ = image.shape
    for start_idx, end_idx in connections:
        pt1 = (int(landmarks[start_idx].x * w), int(landmarks[start_idx].y * h))
        pt2 = (int(landmarks[end_idx].x * w), int(landmarks[end_idx].y * h))
        cv2.line(image, pt1, pt2, color, 2, cv2.LINE_AA)
    for lm in landmarks:
        cv2.circle(image, (int(lm.x * w), int(lm.y * h)), 4, (255, 255, 255), -1)

# PRINT INSTRUCTIONS LAST
print_instructions()

# --- 5. MAIN LOOP ---
cap = cv2.VideoCapture(0)
cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1280)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)

while cap.isOpened():
    success, img = cap.read()
    if not success: break
    img = cv2.flip(img, 1)
    h, w, c = img.shape
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    
    mp_image = mp.Image(image_format=mp.ImageFormat.SRGB, data=img_rgb)
    timestamp = int(time.time() * 1000)
    detection_result = detector.detect_for_video(mp_image, timestamp)

    # UI Overlay
    modes_text = ["NAVIGATION", "SCROLLING", "VOLUME", "GRAB & DRAG"]
    mode_colors = [(255, 0, 255), (0, 255, 255), (255, 128, 0), (0, 255, 0)]
    cv2.rectangle(img, (20, 20), (320, 75), (20, 20, 20), -1)
    cv2.rectangle(img, (20, 20), (320, 75), mode_colors[current_mode], 2)
    cv2.putText(img, f"MODE: {modes_text[current_mode]}", (40, 55), 
                cv2.FONT_HERSHEY_DUPLEX, 0.7, (255, 255, 255), 1, cv2.LINE_AA)

    if detection_result.hand_landmarks:
        lms = detection_result.hand_landmarks[0]
        t_tip, i_base, i_tip = lms[4], lms[5], lms[8]
        m_tip, r_tip, p_tip = lms[12], lms[16], lms[20]
        
        margin = 150 
        tx = np.interp(i_tip.x * w, (margin, w - margin), (0, screen_w))
        ty = np.interp(i_tip.y * h, (margin, h - margin), (0, screen_h))
        currX = plocX + (tx - plocX) / smoothening
        currY = plocY + (ty - plocY) / smoothening

        idx_up = i_tip.y < lms[6].y
        mid_up = m_tip.y < lms[10].y
        rng_up = r_tip.y < lms[14].y
        pnk_up = p_tip.y < lms[18].y
        up_count = [idx_up, mid_up, rng_up, pnk_up].count(True)

        thumb_gap = abs(t_tip.x - i_base.x)
        brake_active_now = (thumb_gap > THUMB_STRETCH_THRESHOLD) and up_count > 0
        if brake_active_now: brake_lock_expiry = time.time() + BRAKE_GRACE_PERIOD
        brake_active = time.time() < brake_lock_expiry

        # --- GRAB PULSE ---
        if up_count == 4: 
            grab_pulse_ready, last_open_time = True, time.time()
            if current_mode != 0: current_mode = 0 
            if dragging: pyautogui.mouseUp(); dragging = False
        if grab_pulse_ready and up_count == 0 and (time.time() - last_open_time) < 0.4:
            current_mode = 3; grab_pulse_ready = False

        # --- MODE SWITCHING ---
        if up_count != last_detected_up_count:
            gesture_start_time, last_detected_up_count = time.time(), up_count
        
        hold_duration = time.time() - gesture_start_time
        if current_mode == 0 and up_count in [2, 3]:
            prog = min(hold_duration / REQUIRED_HOLD_TIME, 1.0)
            cv2.rectangle(img, (40, 90), (240, 105), (30, 30, 30), -1)
            cv2.rectangle(img, (40, 90), (40 + int(prog * 200), 105), (0, 255, 0), -1)
            if prog >= 1.0: current_mode = 1 if up_count == 2 else 2

        # --- EXECUTION ---
        dist_pinch = math.hypot(int(i_tip.x*w) - int(t_tip.x*w), int(i_tip.y*h) - int(t_tip.y*h))
        if current_mode == 0:
            if brake_active:
                cv2.putText(img, "BRAKE LOCKED", (int(i_tip.x*w)+20, int(i_tip.y*h)), 1, 1.2, (0, 255, 255), 2)
                if dist_pinch < 35 and not clicked: pyautogui.click(); clicked = True; brake_lock_expiry = 0
            else:
                if idx_up: pyautogui.moveTo(currX, currY)
                clicked = False
        elif current_mode == 1: pyautogui.scroll(int((lms[0].y - i_tip.y) * 100))
        elif current_mode == 2:
            diff = i_tip.y - lms[0].y
            if diff < -0.2: pyautogui.press("volumeup")
            elif diff > 0.1: pyautogui.press("volumedown")
        elif current_mode == 3:
            if brake_active and dist_pinch < 35:
                if not dragging: pyautogui.mouseDown(); dragging = True
            else:
                pyautogui.moveTo(currX, currY)
                if dragging and dist_pinch > 60: pyautogui.mouseUp(); dragging = False

        line_color = (0, 255, 0) if brake_active_now else (0, 255, 255) if brake_active else (0, 0, 255)
        draw_landmarks_manual(img, lms, line_color)
        plocX, plocY = currX, currY

    cv2.imshow("Precision Gesture Controller", img)
    if cv2.waitKey(1) & 0xFF == ord('q'): break

cap.release()
cv2.destroyAllWindows()
